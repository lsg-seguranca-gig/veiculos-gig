var NOME_ABA_MOVIMENTACOES = 'Movimentacoes'; var NOME_ABA_VEICULOS = 'Cadastros_Veiculos'; var NOME_ABA_FUNCIONARIOS = 'Cadastros_Funcionarios'; var NOME_ABA_CONTROLADORES = 'Cadastros_Controladores'; function doGet(e) { var acao = e ? e.parameter.acao : null; if (acao === 'obterTodos' || acao === 'obterMovimentacoes') { return responderJSON(obterMovimentacoesComOpcoes()); } if (acao === 'obterOpcoes') { return responderJSON(obterOpcoes()); } return responderJSON(obterMovimentacoesComOpcoes()); } function doPost(e) { try { var dados = JSON.parse(e.postData.contents); var acao = dados.acao; if (acao === 'registrarSaida') { return responderJSON(registrarSaida(dados)); } if (acao === 'registrarRetorno') { return responderJSON(registrarRetorno(dados)); } return responderJSON({ status: 'erro', mensagem: 'Ação POST não reconhecida.' }); } catch (err) { return responderJSON({ status: 'erro', mensagem: err.toString() }); } } function registrarSaida(dados) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var aba = ss.getSheetByName(NOME_ABA_MOVIMENTACOES) || ss.getSheetByName('Movimentações') || ss.getSheets()[0]; var tz = ss.getSpreadsheetTimeZone(); var agora = new Date(); var dataHoje = Utilities.formatDate(agora, tz, "dd/MM/yyyy HH:mm"); var proximoId = 1; if (aba.getLastRow() > 1) { var ultimosIds = aba.getRange(2, 1, aba.getLastRow() - 1, 1).getValues(); var maxId = 0; for (var i = 0; i < ultimosIds.length; i++) { var idNum = parseInt(ultimosIds[i][0], 10); if (!isNaN(idNum) && idNum > maxId) { maxId = idNum; } } proximoId = maxId + 1; } var headers = aba.getRange(1, 1, 1, aba.getLastColumn()).getValues()[0].map(function(h) { return normalizarTexto(String(h)); }); var novaLinha = new Array(headers.length).fill(''); function preencherCampo(nomesPossiveis, valor) { for (var i = 0; i < nomesPossiveis.length; i++) { var idx = headers.indexOf(normalizarTexto(nomesPossiveis[i])); if (idx !== -1) { novaLinha[idx] = valor; return; } } } novaLinha[0] = proximoId; preencherCampo(['Data Saida', 'Data/Hora Saida', 'Data Saída', 'Data/Hora Saída', 'Saída', 'Data/Hora'], dataHoje); preencherCampo(['Veiculo', 'Veículo', 'Placa', 'VTR'], dados.veiculo || ''); preencherCampo(['Motorista', 'Nome Motorista'], dados.motorista || ''); preencherCampo(['Auxiliar', 'Nome Auxiliar', 'Auxiliares', 'Ajudante'], dados.auxiliar || dados.auxiliares || ''); preencherCampo(['Controlador Saida', 'Controlador Saída', 'Controlador S', 'Controlador'], dados.controladorSaida || ''); preencherCampo(['Observacao', 'Observação', 'Destino', 'Obs', 'Observacoes'], dados.obsSaida || dados.observacoes || ''); aba.appendRow(novaLinha); return { status: 'sucesso', mensagem: 'Saída registrada com sucesso!', id: proximoId }; } function registrarRetorno(dados) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var aba = ss.getSheetByName(NOME_ABA_MOVIMENTACOES) || ss.getSheetByName('Movimentações') || ss.getSheets()[0]; if (aba.getLastRow() <= 1) { return { status: 'erro', mensagem: 'Nenhum registro encontrado na planilha.' }; } var tz = ss.getSpreadsheetTimeZone(); var agora = new Date(); var dataHoje = Utilities.formatDate(agora, tz, "dd/MM/yyyy HH:mm"); var idProcurado = String(dados.id).trim(); var ctrlRetorno = dados.controladorChegada || dados.controladorRetorno || ''; var obsRetorno = dados.obsRetorno || dados.observacoes || ''; var numCols = aba.getLastColumn(); var todosOsValores = aba.getRange(1, 1, aba.getLastRow(), numCols).getValues(); var headers = todosOsValores[0].map(function(h) { return normalizarTexto(String(h)); }); function getColIndex(nomesPossiveis) { for (var i = 0; i < nomesPossiveis.length; i++) { var idx = headers.indexOf(normalizarTexto(nomesPossiveis[i])); if (idx !== -1) return idx + 1; } return -1; } var colRetornoIndex = getColIndex(['Data Retorno', 'Data Chegada', 'Data/Hora Retorno', 'Retorno', 'Chegada']); var colCtrlRetornoIndex = getColIndex(['Controlador Retorno', 'Controlador Chegada', 'Controlador R']); var colObsIndex = getColIndex(['Observacao', 'Observação', 'Obs', 'Observacoes']); for (var i = 1; i < todosOsValores.length; i++) { if (String(todosOsValores[i][0]).trim() === idProcurado) { var linhaPlanilha = i + 1; if (colRetornoIndex !== -1) { aba.getRange(linhaPlanilha, colRetornoIndex).setValue(dataHoje); } if (colCtrlRetornoIndex !== -1 && ctrlRetorno) { aba.getRange(linhaPlanilha, colCtrlRetornoIndex).setValue(ctrlRetorno); } if (colObsIndex !== -1 && obsRetorno) { var obsAtual = String(todosOsValores[i][colObsIndex - 1] || ''); var novaObs = obsAtual ? (obsAtual + " | Retorno: " + obsRetorno) : obsRetorno; aba.getRange(linhaPlanilha, colObsIndex).setValue(novaObs); } return { status: 'sucesso', mensagem: 'Retorno registrado com sucesso!' }; } } return { status: 'erro', mensagem: 'Registro com ID #' + idProcurado + ' não foi encontrado.' }; } function obterMovimentacoesComOpcoes() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var tz = ss.getSpreadsheetTimeZone(); var opcoesCadastradas = obterOpcoes(); var abaMov = ss.getSheetByName(NOME_ABA_MOVIMENTACOES); if (!abaMov) { abaMov = ss.getSheetByName('Movimentações') || ss.getSheets()[0]; } var movimentacoes = []; if (abaMov && abaMov.getLastRow() > 1) { var numCols = abaMov.getLastColumn(); var todosOsValores = abaMov.getRange(1, 1, abaMov.getLastRow(), numCols).getValues(); var headers = todosOsValores[0].map(function(h) { return normalizarTexto(String(h)); }); var dados = todosOsValores.slice(1); function getVal(linha, nomesPossiveis) { for (var i = 0; i < nomesPossiveis.length; i++) { var busca = normalizarTexto(nomesPossiveis[i]); var idx = headers.indexOf(busca); if (idx !== -1 && linha[idx] !== undefined && linha[idx] !== null && String(linha[idx]).trim() !== '') { return linha[idx]; } } return ''; } function formatarDataHora(val) { if (!val) return { data: '', hora: '' }; if (val instanceof Date) { return { data: Utilities.formatDate(val, tz, "dd/MM/yyyy"), hora: Utilities.formatDate(val, tz, "HH:mm") }; } var str = String(val).trim(); if (str.indexOf('/') !== -1) { var partesEspaco = str.split(' '); var parteData = partesEspaco[0]; var parteHora = partesEspaco[1] || ''; var dP = parteData.split('/'); if (dP.length === 3) { var dia = dP[0].padStart(2, '0'); var mes = dP[1].padStart(2, '0'); var ano = dP[2]; var horaStr = ''; if (parteHora && parteHora.indexOf(':') !== -1) { var hP = parteHora.split(':'); horaStr = hP[0].padStart(2, '0') + ':' + hP[1].padStart(2, '0'); } return { data: dia + '/' + mes + '/' + ano, hora: horaStr }; } } var d = new Date(str); if (!isNaN(d.getTime())) { return { data: Utilities.formatDate(d, tz, "dd/MM/yyyy"), hora: Utilities.formatDate(d, tz, "HH:mm") }; } return { data: str, hora: '' }; } for (var i = 0; i < dados.length; i++) { var linha = dados[i]; var id = linha[0]; if (!id) continue; var rawSaida = getVal(linha, [ 'Data Saida', 'Data/Hora Saida', 'Data Saída', 'Data/Hora Saída', 'Saída', 'Data/Hora', 'Data e Hora de Saida', 'Data e Hora de Saída' ]); var rawRetorno = getVal(linha, [ 'Data Retorno', 'Data Chegada', 'Data/Hora Retorno', 'Retorno', 'Chegada', 'Data e Hora de Retorno', 'Data e Hora de Chegada' ]); var saidaFmt = formatarDataHora(rawSaida); var retornoFmt = formatarDataHora(rawRetorno); var kmSaida = getVal(linha, ['Km Saida', 'KM Saída', 'Km Inicial', 'KM Inicial', 'Odometro Saida', 'Hodometro Saida']); var kmRetorno = getVal(linha, ['Km Retorno', 'KM Retorno', 'Km Final', 'KM Final', 'Odometro Retorno', 'Hodometro Retorno']); var combSaida = getVal(linha, ['Combustivel Saida', 'Combustível Saída', 'Comb. Saída', 'Nível Combustível Saída']); var combRetorno = getVal(linha, ['Combustivel Retorno', 'Combustível Retorno', 'Comb. Retorno', 'Nível Combustível Retorno']); movimentacoes.push({ id: String(id), dataSaida: saidaFmt.data, horarioSaida: saidaFmt.hora, veiculo: getVal(linha, ['Veiculo', 'Veículo', 'Placa', 'VTR']), controladorSaida: getVal(linha, ['Controlador Saida', 'Controlador Saída', 'Controlador S', 'Controlador', 'Controlador de Saída']), motorista: getVal(linha, ['Motorista', 'Nome Motorista', 'Nome do Motorista']), auxiliar: getVal(linha, ['Auxiliar', 'Nome Auxiliar', 'Nome do Auxiliar', 'Auxiliares', 'Ajudante', 'Ajudantes', 'Acompanhante', 'Aux', 'Nome Aux']), dataChegada: retornoFmt.data, horarioChegada: retornoFmt.hora, controladorChegada: getVal(linha, ['Controlador Retorno', 'Controlador Chegada', 'Controlador R', 'Controlador de Retorno']), obsSaida: getVal(linha, ['Observacao', 'Observação', 'Destino', 'Obs', 'Observacoes']), combustivelSaida: String(combSaida), combustivelChegada: String(combRetorno), kmSaida: String(kmSaida), kmChegada: String(kmRetorno), status: (retornoFmt.data && retornoFmt.data !== '-' && retornoFmt.data !== '') ? 'CONCLUIDO' : 'EM_TRANSITO' }); } } return { status: 'sucesso', registros: movimentacoes, veiculos: opcoesCadastradas.veiculos, motoristas: opcoesCadastradas.motoristas, auxiliares: opcoesCadastradas.auxiliares, controladores: opcoesCadastradas.controladores }; } function obterOpcoes() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return { veiculos: lerColunaAba(ss, NOME_ABA_VEICULOS), motoristas: lerColunaAba(ss, NOME_ABA_FUNCIONARIOS), auxiliares: lerColunaAba(ss, NOME_ABA_FUNCIONARIOS), controladores: lerColunaAba(ss, NOME_ABA_CONTROLADORES) }; } function lerColunaAba(ss, nomeAba) { var aba = ss.getSheetByName(nomeAba); if (!aba) { var sheets = ss.getSheets(); for (var i = 0; i < sheets.length; i++) { if (normalizarTexto(sheets[i].getName()) === normalizarTexto(nomeAba)) { aba = sheets[i]; break; } } } if (!aba || aba.getLastRow() <= 1) return []; var valores = aba.getRange(2, 1, aba.getLastRow() - 1, 1).getValues(); var lista = []; for (var i = 0; i < valores.length; i++) { var val = String(valores[i][0]).trim(); if (val !== '' && lista.indexOf(val) === -1) { lista.push(val); } } return lista.sort(); } function normalizarTexto(str) { if (!str) return ''; return str.toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .replace(/[^a-z0-9]/g, ""); } function responderJSON(objeto) { return ContentService .createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); }
